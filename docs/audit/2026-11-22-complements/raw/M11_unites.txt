Type=simple
User=jetson
Environment=ROS_DOMAIN_ID=28
Environment=HOME=/home/jetson

# config/waypoints.yaml, pour que le catalogue et la carte chargee ne puissent pas diverger.

Restart=on-failure
RestartSec=10

# /cmd_vel_safe a zero par son chien de garde de 0,5 s.

TimeoutStopSec=30
